# Signal Engine crontab entries
# Install with: crontab -e (merge these lines)
# Assumes repo checked out at ~/signal-engine with .venv ready.

# Pull new posts + run analysis every 30 minutes.
# fetch is the polite ingestion step; analyze is the deterministic signal
# extraction (phrase stats, intent scoring, pain clustering). Both must
# finish before 08:00 so the morning digest has fresh material.
*/30 * * * *  cd $HOME/signal-engine && . .venv/bin/activate && ./run.sh fetch analyze >> data/cron.log 2>&1

# Morning digest at 08:00 local. The dashboard reads the most recent
# digest row, so this gates the operator's day.
0 8 * * *     cd $HOME/signal-engine && . .venv/bin/activate && ./run.sh digest >> data/cron.log 2>&1

# Weekly cross-subreddit niche report, Mondays 07:00. M3 #13 ships a
# writer that combines precision@10 and the trailing-7-day digest
# trends. Until then, this entry is a no-op (the command is registered
# as not-implemented).
0 7 * * 1     cd $HOME/signal-engine && . .venv/bin/activate && ./run.sh report --weekly >> data/cron.log 2>&1

# Daily health check at 09:00 — surfaces fetch error rate and breaker
# events from the trailing 24h, so quiet cron breaks don't go unnoticed.
0 9 * * *     cd $HOME/signal-engine && . .venv/bin/activate && ./run.sh status >> data/cron.log 2>&1
